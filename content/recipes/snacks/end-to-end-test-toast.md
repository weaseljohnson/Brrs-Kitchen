ingredients:

    ingredients:
      - count: 2 whole
        item: slices sourdough bread
      - count: 1 whole
        item: ripe avocado
      - count: ½ tsp
        item: salt
      - count: ¼ tsp
        item: black pepper
      - item: red pepper flakes, to taste
  - groupName: To Finish
    ingredients:
      - count: 14g (1 tbsp)
        item: extra virgin olive oil
      - count: 1 medium
        item: lemon, juiced
directions:
  - title: Toast the bread
    body: Toast the sourdough slices until golden brown and crisp, about 3–4 minutes.
  - title: Mash the avocado
    body: Halve the avocado, remove the pit, and scoop the flesh into a bowl. Add the lemon juice, salt, and pepper, then mash with a fork until mostly smooth with a few small chunks remaining.
  - title: Assemble
    body: Spread the avocado mixture evenly over the toast. Drizzle with olive oil and finish with a pinch of red pepper flakes.
